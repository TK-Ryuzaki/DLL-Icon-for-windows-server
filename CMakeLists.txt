cmake_minimum_required(VERSION 3.24)
project(ADIcons VERSION 1.0.0 LANGUAGES CXX RC)

if(NOT WIN32)
    message(FATAL_ERROR "ADIcons is a Windows resource DLL and must be built with the Windows SDK/MSVC toolchain.")
endif()

set(ICON_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons")
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
file(GLOB ICON_FILES CONFIGURE_DEPENDS "${ICON_SOURCE_DIR}/*.ico")
list(SORT ICON_FILES)

set(GENERATED_ICON_RC "${GENERATED_DIR}/icons.rc")
set(GENERATED_IDS "${GENERATED_DIR}/resource_ids.h")
file(WRITE "${GENERATED_ICON_RC}" "// Generated by CMake. Do not edit.\n")
file(WRITE "${GENERATED_IDS}" "// Generated by CMake. Include this file in deployment documentation.\n#pragma once\n")

set(RESOURCE_ID 1000)
set(SEEN_SYMBOLS "")
foreach(ICON_FILE IN LISTS ICON_FILES)
    get_filename_component(ICON_NAME "${ICON_FILE}" NAME_WE)
    string(TOUPPER "${ICON_NAME}" ICON_SYMBOL_SUFFIX)
    string(REGEX REPLACE "[^A-Z0-9_]" "_" ICON_SYMBOL_SUFFIX "${ICON_SYMBOL_SUFFIX}")
    if(ICON_SYMBOL_SUFFIX MATCHES "^[0-9]")
        set(ICON_SYMBOL_SUFFIX "_${ICON_SYMBOL_SUFFIX}")
    endif()
    set(ICON_SYMBOL "ICON_${ICON_SYMBOL_SUFFIX}")
    if(ICON_SYMBOL IN_LIST SEEN_SYMBOLS)
        message(FATAL_ERROR "Icon filenames produce a duplicate resource symbol: ${ICON_NAME}")
    endif()
    list(APPEND SEEN_SYMBOLS "${ICON_SYMBOL}")
    file(APPEND "${GENERATED_ICON_RC}" "${RESOURCE_ID} ICON \"${ICON_FILE}\"\n")
    file(APPEND "${GENERATED_IDS}" "#define ${ICON_SYMBOL} ${RESOURCE_ID}\n")
    math(EXPR RESOURCE_ID "${RESOURCE_ID} + 1")
endforeach()

if(NOT ICON_FILES)
    message(WARNING "No .ico files found in assets/icons. The DLL will contain version metadata only until icons are added.")
endif()

add_library(ADIcons SHARED src/dllmain.cpp resources/version.rc "${GENERATED_ICON_RC}")
target_compile_features(ADIcons PRIVATE cxx_std_17)
target_include_directories(ADIcons PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(ADIcons PROPERTIES
    OUTPUT_NAME "AD-Icons"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)
